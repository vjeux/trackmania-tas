/**
 *	LowG library -- the state and logic behind the LowG mode (Modes/TrackMania/LowG.Script.txt).
 *
 *	A mode that #Extends another script cannot own top-level variables (the compiler
 *	stops with "unexpected end of file"), so everything with state lives here, as
 *	Nadeo's and FlagRush's libraries do.
 *
 *	#RequireContext CSmMode
 */
#Include "TextLib" as TL
#Include "MathLib" as ML

#Const Version "2026-09-26"
#Const ScriptName "Libs/LowG/LowG.Script.txt"

#Const C_Prefix "$0cf[low-g]$z "
#Const C_Event_Cmd "LowG.Cmd"			///< TriggerModeScriptEventArray("LowG.Cmd", [Login, Command, Args...])
#Const C_Event_Status "LowG.Status"	///< sent back with one Text per line
#Const C_BalloonName "$fa0Ballon"		///< nickname prefix of the fake-user balloons
#Const C_ReactorRefreshMs 2000			///< the engine gives no readback of a reactor order: re-issue this often to keep it on

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Globals
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
declare Real[Text] G_GravityByLogin;	///< login => coefficient chosen in chat
declare Real G_DefaultGravity;			///< for players without a choice
declare Real G_BotGravity;
declare Real G_BotAdherence;			///< grip of the balloons: 0 = they slide like on ice
declare Real G_SettingBotGravity;		///< last seen value of S_LowG_BotGravity (change detection)
declare Real G_SettingBotAdherence;	///< last seen value of S_LowG_BotAdherence
declare Boolean G_Collisions;
declare Integer G_WantedBots;
declare Integer G_MaxBots;
declare Boolean G_BotsAreFakeUsers;
declare Integer G_LastBotSync;
declare CUILayer G_HintLayer;
declare Ident[Text] G_BotModels;			///< "sport"|"snow"|"rally"|"desert"|"skin" => item-list Ident for CreateBotPlayer
declare Text G_BotModel;					///< which of them new balloons get ("" = engine default via SetNbBotPlayers)
declare Text G_BotSkin;					///< skin (zip URL or Skins/Models/... path) registered as "skin"
declare Text G_SettingBotSkin;			///< last seen value of S_LowG_BotSkin (change detection)
declare Text G_Reactor;					///< players' permanent reactor: "off" | "down" | "up"
declare Integer G_ReactorLvl;				///< 1 = ReactorBoost, 2 = ReactorBoost2
declare Text G_SettingReactor;
declare Integer G_SettingReactorLvl;
declare Boolean G_ItemListOk;				///< the one-line command reminder at the top of every screen

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //

Text Fmt(Real _Coef) {
	return TL::FormatReal(_Coef, 2, False, False);
}

/// Chat line to one player (Null-safe; bots have no UI).
Void Tell(CSmPlayer _Player, Text _Message) {
	if (_Player == Null) return;
	declare CUIConfig UI <=> UIManager.GetUI(_Player);
	if (UI == Null) return;
	UI.SendChat(C_Prefix ^ _Message);
}

/// Chat line to everyone.
Void TellAll(Text _Message) {
	UIManager.UIAll.SendChat(C_Prefix ^ _Message);
}

Text NameOf(CSmPlayer _Player, Text _Fallback) {
	if (_Player != Null && _Player.User != Null) return _Player.User.Name;
	return _Fallback;
}

/// Only the fake users this library created (by name): an admin's ConnectFakePlayer is left alone.
Boolean IsBalloonUser(CUser _User) {
	return _User.IsFakeUser && TL::StartsWith(C_BalloonName, _User.Name, False, True);
}

/// Engine bots, and the fake users we named "Ballon N". An admin's ConnectFakePlayer is a player.
Boolean IsBalloon(CSmPlayer _Player) {
	if (_Player.IsBot) return True;
	return _Player.IsFakePlayer && _Player.User != Null && IsBalloonUser(_Player.User);
}

/// The coefficient a player should be driving with right now.
Real WantedGravity(CSmPlayer _Player) {
	if (IsBalloon(_Player)) return G_BotGravity;
	if (_Player.User != Null && G_GravityByLogin.existskey(_Player.User.Login)) {
		return G_GravityByLogin[_Player.User.Login];
	}
	return G_DefaultGravity;
}

/**
 *	Push the wanted coefficient to a car. The Delayed setters act on the live vehicle
 *	(that is how FlagRush tunes its cars), but the engine keeps only a few pending
 *	orders per player, so try again next tick when the queue is full. A spawn (StartTime
 *	changes) gives a fresh car, so the order is repeated then.
 */
Void Apply(CSmPlayer _Player) {
	declare Real LowG_AppliedCoef for _Player = -1.;
	declare Real LowG_AppliedAdherence for _Player = -1.;
	declare Integer LowG_AppliedStartTime for _Player = -1;
	declare Integer LowG_AppliedAt for _Player = -1;

	if (_Player.SpawnStatus != CSmPlayer::ESpawnStatus::Spawned) {
		LowG_AppliedStartTime = -1;
		return;
	}
	declare Boolean Balloon = IsBalloon(_Player);
	declare Real Wanted = ML::Clamp(WantedGravity(_Player), 0., 1.);
	// Players keep their normal tyres; balloons get the grip asked for (0 = they slide).
	declare Real WantedAdherence = 1.;
	if (Balloon) WantedAdherence = ML::Clamp(G_BotAdherence, 0., 1.);
	declare Boolean Fresh = (LowG_AppliedStartTime != _Player.StartTime);
	declare Boolean Changed = (LowG_AppliedCoef != Wanted) || (Balloon && LowG_AppliedAdherence != WantedAdherence);
	// The car reports the coefficients it runs with (GravityCoef / AdherenceCoef read back what
	// the Delayed orders set). A checkpoint respawn or an engine reset may put them back to 1:
	// whenever a readback disagrees for more than a second, order again (at most once a second).
	declare Boolean Confirm = (LowG_AppliedAt > 0 && Now - LowG_AppliedAt > 1000 && (_Player.GravityCoef != Wanted || (Balloon && _Player.AdherenceCoef != WantedAdherence)));
	if (!Fresh && !Changed && !Confirm) return;
	if (SetPlayer_DelayedIsFull(_Player)) return;

	SetPlayer_Delayed_GravityCoef(_Player, Wanted);
	// The welcome line goes out on the first spawn: at OnPlayerAdded the player has no UI yet.
	declare Boolean LowG_Welcomed for _Player = False;
	if (!LowG_Welcomed && !Balloon) {
		LowG_Welcomed = True;
		Tell(_Player, "low gravity server: yours is " ^ Fmt(Wanted) ^ " -- /gravity <0..1> to change it, /bots <n> parks balloons, /grip and /botgravity tune them, /collisions on|off, /status");
	}
	if (Balloon) {
		// Balloons must not drive away...
		if (Fresh && !SetPlayer_DelayedIsFull(_Player)) {
			SetPlayer_Delayed_NoEngine(_Player, True);
		}
		// ...and slide with as little grip as asked.
		if (!SetPlayer_DelayedIsFull(_Player)) {
			SetPlayer_Delayed_AdherenceCoef(_Player, WantedAdherence);
			LowG_AppliedAdherence = WantedAdherence;
		}
	}
	declare Integer LowG_ApplyCount for _Player = 0;
	LowG_ApplyCount += 1;
	LowG_AppliedCoef = Wanted;
	LowG_AppliedStartTime = _Player.StartTime;
	LowG_AppliedAt = Now;
}

/**
 *	Players (never balloons) keep a permanent reactor boost: BoostDown/BoostUp (level 1) or
 *	Boost2Down/Boost2Up (level 2), doc.h 2016-2028, "Activate or Deactivate ... with a 250ms
 *	delay". CSmPlayer exposes no reactor readback, so the order is repeated every
 *	C_ReactorRefreshMs and on every spawn; switching to off deactivates all four once.
 */
Void ApplyReactor(CSmPlayer _Player) {
	if (IsBalloon(_Player)) return;
	declare Text LowG_ReactorApplied for _Player = "";
	declare Integer LowG_ReactorAppliedAt for _Player = -1;
	declare Integer LowG_ReactorStartTime for _Player = -1;
	if (_Player.SpawnStatus != CSmPlayer::ESpawnStatus::Spawned) {
		LowG_ReactorStartTime = -1;
		LowG_ReactorApplied = "";
		return;
	}
	declare Text Wanted = G_Reactor ^ G_ReactorLvl;
	if (G_Reactor == "off") Wanted = "off";
	declare Boolean Fresh = (LowG_ReactorStartTime != _Player.StartTime);
	declare Boolean Changed = (LowG_ReactorApplied != Wanted);
	declare Boolean Refresh = (Wanted != "off" && Now - LowG_ReactorAppliedAt > C_ReactorRefreshMs);
	if (!Fresh && !Changed && !Refresh) return;
	if (Wanted == "off") {
		if (LowG_ReactorApplied == "off" && !Fresh) return;
		if (SetPlayer_DelayedIsFull(_Player)) return;
		SetPlayer_Delayed_BoostDown(_Player, False);
		if (!SetPlayer_DelayedIsFull(_Player)) SetPlayer_Delayed_BoostUp(_Player, False);
		if (!SetPlayer_DelayedIsFull(_Player)) SetPlayer_Delayed_Boost2Down(_Player, False);
		if (!SetPlayer_DelayedIsFull(_Player)) SetPlayer_Delayed_Boost2Up(_Player, False);
	} else {
		if (SetPlayer_DelayedIsFull(_Player)) return;
		if (G_Reactor == "down" && G_ReactorLvl >= 2) SetPlayer_Delayed_Boost2Down(_Player, True);
		else if (G_Reactor == "down") SetPlayer_Delayed_BoostDown(_Player, True);
		else if (G_Reactor == "up" && G_ReactorLvl >= 2) SetPlayer_Delayed_Boost2Up(_Player, True);
		else if (G_Reactor == "up") SetPlayer_Delayed_BoostUp(_Player, True);
	}
	LowG_ReactorApplied = Wanted;
	LowG_ReactorAppliedAt = Now;
	LowG_ReactorStartTime = _Player.StartTime;
}

Void ApplyCollisions() {
	if (UsePvPCollisions != G_Collisions) UsePvPCollisions = G_Collisions;
	if (!G_Collisions) return;
	foreach (Player in Players) {
		// Time Attack lets each client simulate itself (nobody interacts). With collisions
		// the server must own the simulation, as FlagRush does.
		if (Player.TrustClientSimu) Player.TrustClientSimu = False;
		if (Player.UseCrudeExtrapolation) Player.UseCrudeExtrapolation = False;
	}
}

Integer BalloonCount() {
	declare Integer N = 0;
	foreach (Player in Players) {
		if (IsBalloon(Player)) N += 1;
	}
	return N;
}

Integer FakeUserCount() {
	declare Integer Have = 0;
	foreach (User in Users) {
		if (IsBalloonUser(User)) Have += 1;
	}
	return Have;
}

Void DestroyBalloonUsers() {
	declare CUser[] Gone;
	foreach (User in Users) {
		if (IsBalloonUser(User)) Gone.add(User);
	}
	foreach (User in Gone) {
		Users_DestroyFake(User);
	}
}

/**
 *	Make the number of balloons match G_WantedBots.
 *	Fake users (default) join like players, so Time Attack itself puts them on the start
 *	line and they show in the player list as "Ballon N". Engine bots (SetNbBotPlayers)
 *	live in BotPlayers only, outside Time Attack's spawn loop, so SpawnBots() below parks
 *	them on the first start block.
 */
Void SyncBots() {
	declare Integer Want = ML::Clamp(G_WantedBots, 0, G_MaxBots);
	if (G_BotsAreFakeUsers) {
		if (BotPlayers.count > 0) DestroyAllBotPlayers();
		declare Integer Have = FakeUserCount();
		if (Have == Want) return;
		if (Want < Have) {
			DestroyBalloonUsers();
			Have = 0;
		}
		for (I, Have + 1, Want) {
			declare CUser Fake <=> Users_CreateFake(C_BalloonName ^ " " ^ I, 0);
		}
	} else {
		if (FakeUserCount() > 0) DestroyBalloonUsers();
		if (BotPlayers.count == Want) return;
		if (Want == 0) {
			DestroyAllBotPlayers();
			return;
		}
		if (G_BotModel != "" && G_BotModels.existskey(G_BotModel) && G_BotModels[G_BotModel] != NullId) {
			// A chosen vehicle/skin: bots are created one by one with that model.
			if (BotPlayers.count > Want) {
				DestroyAllBotPlayers();
			}
			for (I, BotPlayers.count + 1, Want) {
				declare CSmPlayer Bot <=> CreateBotPlayer(G_BotModels[G_BotModel], 0);
			}
		} else {
			SetNbBotPlayers(Want, 0);
		}
	}
}

/// Engine bots are not spawned by anyone else: park the idle ones on the start block.
Void SpawnBots() {
	if (MapLandmarks_PlayerSpawn.count == 0) return;
	foreach (Bot in BotPlayers) {
		if (Bot.SpawnStatus != CSmPlayer::ESpawnStatus::NotSpawned) continue;
		declare Integer LowG_BotSpawnRequestedAt for Bot = -1;
		if (LowG_BotSpawnRequestedAt > 0 && Now - LowG_BotSpawnRequestedAt < 3000) continue;
		LowG_BotSpawnRequestedAt = Now;
		SpawnBotPlayer(Bot, 0, 100, MapLandmarks_PlayerSpawn[0].PlayerSpawn, Now);
	}
}

/**
 *	Register the vehicle models (and one skinned CarSport) the balloons may use. The item list is
 *	the engine's registry of spawnable models; FlagRush registers "CarSport"/"CarSnow"/"CarRally"/
 *	"CarDesert" and skinned variants ("CarSport" + a skin URL) this way and sets ForceModelId.
 *	Whether a second ItemList_Begin() is accepted after the Time Attack base built its own list is
 *	what the harness test establishes (G_ItemListOk + the Idents are logged).
 */
Void RegisterBotModels() {
	G_BotModels.clear();
	ItemList_Begin();
	G_ItemListOk = True;
	G_BotModels["sport"] = ItemList_Add("CarSport");
	G_BotModels["snow"] = ItemList_Add("CarSnow");
	G_BotModels["rally"] = ItemList_Add("CarRally");
	G_BotModels["desert"] = ItemList_Add("CarDesert");
	if (G_BotSkin != "" && G_BotSkin != "default") {
		G_BotModels["skin"] = ItemList_AddWithSkin("CarSport", G_BotSkin);
	}
	ItemList_End();
	declare Text SkinId = "-";
	if (G_BotModels.existskey("skin")) SkinId = "" ^ G_BotModels["skin"];
	log("LowG item list: sport=" ^ G_BotModels["sport"] ^ " snow=" ^ G_BotModels["snow"] ^ " rally=" ^ G_BotModels["rally"] ^ " desert=" ^ G_BotModels["desert"] ^ " skin=" ^ SkinId ^ " (" ^ G_BotSkin ^ ")");
}

/**
 *	The item list can only be built in Match_StartServer: on this server (2026-05-16) an
 *	ItemList_Begin() during play OR at Match_StartMap halts the script ("The procedure
 *	ItemList_Begin failed to complete") and the mode reloads. So the balloon skin is fixed at
 *	start from S_LowG_BotSkin; changing it needs `tmserverctl restart`.
 */
Boolean RegisterBotSkin(Text _Skin) {
	G_BotSkin = _Skin;
	if (_Skin == "default") G_BotSkin = "";
	RegisterBotModels();
	return G_BotModels.existskey("skin") && G_BotModels["skin"] != NullId;
}
/// Snapshot for tmserverctl / debugging: one callback with one Text per line.
Void SendStatus() {
	declare Text[] Lines;
	Lines.add("default=" ^ Fmt(G_DefaultGravity) ^ " collisions=" ^ G_Collisions ^ " balloons=" ^ BalloonCount() ^ "/" ^ G_WantedBots ^ " botgravity=" ^ Fmt(G_BotGravity) ^ " botadherence=" ^ Fmt(G_BotAdherence) ^ " botmodel=" ^ G_BotModel ^ " botskin=" ^ G_BotSkin ^ " reactor=" ^ G_Reactor ^ G_ReactorLvl ^ " fakeusers=" ^ G_BotsAreFakeUsers ^ " players=" ^ Players.count ^ " bots=" ^ BotPlayers.count);
	declare CSmPlayer[] Everyone;
	foreach (Player in Players) Everyone.add(Player);
	foreach (Bot in BotPlayers) {
		if (!Everyone.exists(Bot)) Everyone.add(Bot);
	}
	foreach (Player in Everyone) {
		declare Text Login = "";
		if (Player.User != Null) Login = Player.User.Login;
		declare Integer LowG_ApplyCount for Player = 0;
		Lines.add(Login ^ " spawn=" ^ Player.SpawnStatus ^ " gravity=" ^ Fmt(Player.GravityCoef) ^ " wanted=" ^ Fmt(WantedGravity(Player)) ^ " adherence=" ^ Fmt(Player.AdherenceCoef) ^ " balloon=" ^ IsBalloon(Player) ^ " model=" ^ Player.ForceModelId ^ " orders=" ^ LowG_ApplyCount ^ " kmh=" ^ ML::NearestInteger(Player.Speed * 3.6) ^ " y=" ^ TL::FormatReal(Player.Position.Y, 1, False, False) ^ " air=" ^ Player.IsInAir);
	}
	XmlRpc.SendCallbackArray(C_Event_Status, Lines);
}

Void HandleCommand(Text[] _Params) {
	if (_Params.count < 2) return;
	declare Text Login = _Params[0];
	declare Text Cmd = TL::ToLowerCase(_Params[1]);
	declare CSmPlayer Caller <=> GetPlayer(Login);

	switch (Cmd) {
		case "gravity": {
			if (_Params.count < 3) {
				declare Text Current = "server default " ^ Fmt(G_DefaultGravity);
				if (G_GravityByLogin.existskey(Login)) Current = "yours " ^ Fmt(G_GravityByLogin[Login]);
				Tell(Caller, "gravity: " ^ Current ^ " -- /gravity <0..1>, /gravity all <0..1>, /gravity reset");
				return;
			}
			declare Text Arg = TL::ToLowerCase(_Params[2]);
			if (Arg == "reset") {
				G_GravityByLogin.removekey(Login);
				Tell(Caller, "gravity back to the server default " ^ Fmt(G_DefaultGravity));
				return;
			}
			if (Arg == "all") {
				if (_Params.count < 4) {
					Tell(Caller, "usage: /gravity all <0..1>");
					return;
				}
				declare Real CoefAll = ML::Clamp(TL::ToReal(_Params[3]), 0., 1.);
				G_DefaultGravity = CoefAll;
				G_GravityByLogin.clear();
				TellAll(NameOf(Caller, Login) ^ "$z set everyone's gravity to " ^ Fmt(CoefAll));
				return;
			}
			declare Real Coef = ML::Clamp(TL::ToReal(Arg), 0., 1.);
			G_GravityByLogin[Login] = Coef;
			Tell(Caller, "your gravity is now " ^ Fmt(Coef) ^ " (1 = normal, 0 = weightless)");
		}
		case "bots": {
			if (_Params.count < 3) {
				Tell(Caller, "balloons: " ^ BalloonCount() ^ " -- /bots <0.." ^ G_MaxBots ^ ">");
				return;
			}
			declare Integer N = ML::Clamp(TL::ToInteger(_Params[2]), 0, G_MaxBots);
			if (_Params.count >= 4) {
				declare Text Kind = TL::ToLowerCase(_Params[3]);
				if (Kind == "fake" || Kind == "fakes" || Kind == "user" || Kind == "users") G_BotsAreFakeUsers = True;
				if (Kind == "bot" || Kind == "bots" || Kind == "engine") G_BotsAreFakeUsers = False;
			}
			G_WantedBots = N;
			SyncBots();
			TellAll(NameOf(Caller, Login) ^ "$z asked for " ^ N ^ " balloon(s)");
		}
		case "grip": {
			if (_Params.count < 3) {
				Tell(Caller, "balloon grip: " ^ Fmt(G_BotAdherence) ^ " -- /grip <0..1> (0 = no friction, 1 = normal tyres)");
				return;
			}
			G_BotAdherence = ML::Clamp(TL::ToReal(_Params[2]), 0., 1.);
			TellAll(NameOf(Caller, Login) ^ "$z set the balloons' grip to " ^ Fmt(G_BotAdherence));
		}
		case "botgravity": {
			if (_Params.count < 3) {
				Tell(Caller, "balloon gravity: " ^ Fmt(G_BotGravity) ^ " -- /botgravity <0..1>");
				return;
			}
			G_BotGravity = ML::Clamp(TL::ToReal(_Params[2]), 0., 1.);
			TellAll(NameOf(Caller, Login) ^ "$z set the balloons' gravity to " ^ Fmt(G_BotGravity));
		}
		case "botmodel": {
			if (_Params.count < 3) {
				declare Text Current = G_BotModel;
				if (Current == "") Current = "default";
				Tell(Caller, "balloon model: " ^ Current ^ " -- /botmodel sport|snow|rally|desert|skin|default (item list ok=" ^ G_ItemListOk ^ ")");
				return;
			}
			declare Text Model = TL::ToLowerCase(_Params[2]);
			if (Model == "default" || Model == "reset") {
				G_BotModel = "";
			} else if (!G_BotModels.existskey(Model) || G_BotModels[Model] == NullId) {
				Tell(Caller, "unknown or unregistered model " ^ Model ^ " -- sport|snow|rally|desert|skin|default");
				return;
			} else {
				G_BotModel = Model;
			}
			if (BotPlayers.count > 0) DestroyAllBotPlayers();
			SyncBots();
			TellAll(NameOf(Caller, Login) ^ "$z set the balloons' model to " ^ Model);
		}
		case "skin": {
			declare Text Current = G_BotSkin;
			if (Current == "") Current = "default livery";
			if (_Params.count >= 3 && _Params[2] != Current) {
				Tell(Caller, "the balloon skin is fixed when the server starts (S_LowG_BotSkin in the matchsettings + restart); now: " ^ Current);
			} else {
				Tell(Caller, "balloon skin: " ^ Current ^ " (change: S_LowG_BotSkin + restart; /botmodel skin|sport|snow|rally|desert switches the model)");
			}
		}
		case "reactor": {
			if (_Params.count < 3) {
				Tell(Caller, "players' reactor: " ^ G_Reactor ^ " level " ^ G_ReactorLvl ^ " -- /reactor down|up|off [1|2]");
				return;
			}
			declare Text Mode = TL::ToLowerCase(_Params[2]);
			if (Mode != "down" && Mode != "up" && Mode != "off") {
				Tell(Caller, "usage: /reactor down|up|off [1|2]");
				return;
			}
			G_Reactor = Mode;
			if (_Params.count >= 4) G_ReactorLvl = ML::Clamp(TL::ToInteger(_Params[3]), 1, 2);
			TellAll(NameOf(Caller, Login) ^ "$z set the players' reactor to " ^ G_Reactor ^ " (level " ^ G_ReactorLvl ^ ")");
		}
		case "collisions": {
			if (_Params.count < 3) {
				Tell(Caller, "collisions: " ^ G_Collisions ^ " -- /collisions on|off");
				return;
			}
			declare Text Arg = TL::ToLowerCase(_Params[2]);
			G_Collisions = (Arg == "on" || Arg == "1" || Arg == "true" || Arg == "yes");
			ApplyCollisions();
			TellAll(NameOf(Caller, Login) ^ "$z turned collisions " ^ Arg);
		}
		case "status": {
			Tell(Caller, "gravity default " ^ Fmt(G_DefaultGravity) ^ ", collisions " ^ G_Collisions ^ ", balloons " ^ BalloonCount() ^ " (gravity " ^ Fmt(G_BotGravity) ^ ", grip " ^ Fmt(G_BotAdherence) ^ "), players " ^ Players.count);
			SendStatus();
		}
		default: {
			Tell(Caller, "commands: /gravity <0..1> | /gravity all <0..1> | /gravity reset | /bots <n> | /grip <0..1> | /botgravity <0..1> | /botmodel sport|snow|rally|desert|skin | /reactor down|up|off [1|2] | /collisions on|off | /status");
		}
	}
}

Void HandleEvents() {
	foreach (Event in XmlRpc.PendingEvents) {
		if (Event.Type != CXmlRpcEvent::EType::CallbackArray) continue;
		if (Event.ParamArray1 == C_Event_Cmd) {
			HandleCommand(Event.ParamArray2);
		} else if (Event.ParamArray1 == C_Event_Status) {
			SendStatus();
		}
	}
}


/// A permanent one-line reminder of the chat commands (a chat message is gone in a minute).
Void CreateHint() {
	if (G_HintLayer != Null) return;
	G_HintLayer <=> UIManager.UILayerCreate();
	G_HintLayer.ManialinkPage = """<manialink version="3" name="LowG_Hint">
	<frame pos="0 85" z-index="20">
		<quad pos="0 0" size="190 7" halign="center" valign="center" bgcolor="00000099" z-index="0"/>
		<label pos="0 0" size="186 6" halign="center" valign="center2" textsize="1.3" textcolor="FFFFFF" textemboss="1" text="$0cfLOW-G$fff  chat:  /gravity 0..1 (toi)   /gravity all 0.3   /bots 4 (ballons)   /grip 0..1   /botgravity 0..1   /collisions on|off   /status" z-index="1"/>
	</frame>
</manialink>""";
	UIManager.UIAll.UILayers.add(G_HintLayer);
}

Void DestroyHint() {
	if (G_HintLayer == Null) return;
	UIManager.UIAll.UILayers.remove(G_HintLayer);
	UIManager.UILayerDestroy(G_HintLayer);
	G_HintLayer <=> Null;
}

/// Match_StartServer: take the settings, reset the state.
Void Load(Real _DefaultGravity, Boolean _Collisions, Integer _Bots, Real _BotGravity, Real _BotAdherence, Integer _MaxBots, Boolean _BotsAreFakeUsers, Text _BotSkin, Text _Reactor, Integer _ReactorLvl) {
	G_Reactor = TL::ToLowerCase(_Reactor);
	if (G_Reactor != "down" && G_Reactor != "up") G_Reactor = "off";
	G_ReactorLvl = ML::Clamp(_ReactorLvl, 1, 2);
	G_SettingReactor = _Reactor;
	G_SettingReactorLvl = _ReactorLvl;
	G_DefaultGravity = ML::Clamp(_DefaultGravity, 0., 1.);
	G_BotAdherence = ML::Clamp(_BotAdherence, 0., 1.);
	G_SettingBotGravity = _BotGravity;
	G_SettingBotAdherence = _BotAdherence;
	G_Collisions = _Collisions;
	G_WantedBots = _Bots;
	G_BotGravity = ML::Clamp(_BotGravity, 0., 1.);
	G_MaxBots = _MaxBots;
	G_BotsAreFakeUsers = _BotsAreFakeUsers;
	G_GravityByLogin.clear();
	G_LastBotSync = 0;
	G_BotModel = "";
	G_SettingBotSkin = _BotSkin;
	if (RegisterBotSkin(_BotSkin)) G_BotModel = "skin";
	// Low-gravity times must not pollute the normal leaderboards.
	Scores_AutoUploadPersonalBests = False;
	SetGameModeNameAndCustomData("LowG", "", "");
	ApplyCollisions();
	CreateHint();
	log("LowG " ^ Version ^ " loaded: default gravity " ^ G_DefaultGravity ^ ", collisions " ^ G_Collisions ^ ", bots " ^ G_WantedBots);
}

/// Settings may change at runtime (SetModeScriptSettings); the mode passes them every tick.
/// A setting is applied when ITS value changes, so a value set in chat (/grip, /botgravity)
/// holds until the setting itself is edited. (The balloon kind is only taken at load time:
/// /bots N fake|bot changes it live.)
Void UpdateSettings(Real _BotGravity, Real _BotAdherence, Integer _MaxBots, Text _BotSkin, Text _Reactor, Integer _ReactorLvl) {
	if (_Reactor != G_SettingReactor) {
		G_SettingReactor = _Reactor;
		G_Reactor = TL::ToLowerCase(_Reactor);
		if (G_Reactor != "down" && G_Reactor != "up") G_Reactor = "off";
	}
	if (_ReactorLvl != G_SettingReactorLvl) {
		G_SettingReactorLvl = _ReactorLvl;
		G_ReactorLvl = ML::Clamp(_ReactorLvl, 1, 2);
	}
	if (_BotSkin != G_SettingBotSkin) {
		// Cannot rebuild the item list now (see RegisterBotSkin): say so once, apply at the next start.
		G_SettingBotSkin = _BotSkin;
		log("LowG: S_LowG_BotSkin changed to " ^ _BotSkin ^ " -- takes effect after a server restart");
		TellAll("balloon skin setting changed; it applies after the next server restart");
	}
	if (_BotGravity != G_SettingBotGravity) {
		G_SettingBotGravity = _BotGravity;
		G_BotGravity = ML::Clamp(_BotGravity, 0., 1.);
	}
	if (_BotAdherence != G_SettingBotAdherence) {
		G_SettingBotAdherence = _BotAdherence;
		G_BotAdherence = ML::Clamp(_BotAdherence, 0., 1.);
	}
	G_MaxBots = _MaxBots;
}

/// Match_StartMap
Void StartMap() {
	// Time Attack may set these per map; low-gravity times stay off the real leaderboards.
	Scores_AutoUploadPersonalBests = False;
	SetGameModeNameAndCustomData("LowG", "", "");
	ApplyCollisions();
	SyncBots();
}

/// Match_Yield
Void Yield() {
	HandleEvents();
	ApplyCollisions();
	foreach (Player in Players) {
		Apply(Player);
		ApplyReactor(Player);
	}
	foreach (Bot in BotPlayers) {
		Apply(Bot);
	}
	SpawnBots();
	if (G_WantedBots > 0 && Now - G_LastBotSync > 5000) {
		G_LastBotSync = Now;
		SyncBots();
	}
}

Void Unload() {
	DestroyHint();
	G_GravityByLogin.clear();
}
